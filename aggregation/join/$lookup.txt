$lookup : 2개 이상의 컬렉션을 JOIN 시키는 연산자

ex)

-- TARGET --

// User
[
  { _id: ObjectId(..1), name: "Alice" },
  { _id: ObjectId(..2), name: "Bob" },
  { _id: ObjectId(..3), name: "Charlie" }
]

// Post
[
  { _id: ObjectId(..a), title: "Post 1", content: "This is post 1.", author:  ObjectId(..1) },
  { _id: ObjectId(..b), title: "Post 2", content: "This is post 2.", author: ObjectId(..2) },
  { _id: ObjectId(..c), title: "Post 3", content: "This is post 3.", author: ObjectId(..3) }
]

-- EXECUTION --

Post.aggregate([
  {
    $lookup: {
      from: "users", // users 컬렉션과 JOIN
      localField: "author", // posts 컬렉션의 author 필드
      foreignField: "_id", // users 컬렉션의 _id 필드
      as: "author" // 결과 필드 이름
    }
  }
])

-- RESULT --

[
  { 
    _id: ObjectId(..a), 
    title: "Post 1", 
    content: "This is post 1.",
    author: [{ _id: ObjectId(..1), name: "Alice" }]
  },
  { 
    _id: ObjectId(..b), 
    title: "Post 2", 
    content: "This is post 2.",
    author: [{_id: ObjectId(..2), name: "Bob" }]
  },
  { 
    _id: ObjectId(..c), 
    title: "Post 3", 
    content: "This is post 3.",
    author: [{ _id: ObjectId(..3), name: "Charlie" }]
  }
]
